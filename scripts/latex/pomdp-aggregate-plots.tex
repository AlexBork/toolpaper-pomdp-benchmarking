\newcommand{\pomdpplot@allpalette}{%
  \pomdpplot@get\pomdpplot@series{\pomdpplot@row}{instance_index}\pomdpplot@seriesindex
  % Six colours and five shapes give 30 distinct colour/marker combinations.
  \pgfmathtruncatemacro\pomdpplot@colourindex{mod(\pomdpplot@seriesindex,6)}%
  \edef\pomdpplot@colour{pomdpplotseries\pomdpplot@colourindex}%
  \pgfmathtruncatemacro\pomdpplot@markindex{mod(\pomdpplot@seriesindex,5)}%
  \ifcase\pomdpplot@markindex\def\pomdpplot@mark{o}%
  \or\def\pomdpplot@mark{square}%
  \or\def\pomdpplot@mark{triangle}%
  \or\def\pomdpplot@mark{diamond}%
  \or\def\pomdpplot@mark{pentagon}%
  \fi
}
\newcommand{\pomdpplot@failuretick}[2]{%
  \pomdpplot@get\pomdpplot@series{0}{has_#1_line}\pomdpplot@hasline
  \ifnum\pomdpplot@hasline=1\relax
    \pomdpplot@get\pomdpplot@series{0}{\pomdpplot@failuremetric _#1_y}\pomdpplot@level
    \edef\pomdpplot@ticks{\pomdpplot@ticks,\pomdpplot@level}%
    \edef\pomdpplot@extraticks{\pomdpplot@extraticks\pomdpplot@extraseparator\pomdpplot@level}%
    \def\pomdpplot@extraseparator{,}%
    \edef\pomdpplot@ticklabels{\unexpanded\expandafter{\pomdpplot@ticklabels},{#2}}%
  \fi
}
\newcommand{\pomdpplot@failureaxis}[1]{%
  \def\pomdpplot@failuremetric{#1}%
  \pomdpplot@get\pomdpplot@series{0}{#1_ymin}\pomdpplot@ymin
  \pomdpplot@get\pomdpplot@series{0}{#1_ymax}\pomdpplot@ymax
  \def\pomdpplot@ticks{}\def\pomdpplot@ticklabels{}\def\pomdpplot@separator{}%
  \def\pomdpplot@extraticks{}\def\pomdpplot@extraseparator{}%
  \def\pomdpplot@runtime{runtime}\def\pomdpplot@metric{#1}%
  \ifx\pomdpplot@metric\pomdpplot@runtime
    \pomdpplot@get\pomdpplot@series{0}{runtime_tick_min}\pomdpplot@tickmin
    \pomdpplot@get\pomdpplot@series{0}{runtime_tick_max}\pomdpplot@tickmax
    \pgfplotsforeachungrouped \pomdpplot@index in {\pomdpplot@tickmin,...,\pomdpplot@tickmax}{%
      \edef\pomdpplot@ticks{\pomdpplot@ticks\pomdpplot@separator 1e\pomdpplot@index}%
      \ifnum\pomdpplot@index=3\relax
        \edef\pomdpplot@ticklabels{\pomdpplot@ticklabels\pomdpplot@separator {$10^{3}$}}%
      \else
        \edef\pomdpplot@ticklabels{\pomdpplot@ticklabels\pomdpplot@separator
          {\noexpand\pgfmathprintnumber[fixed,precision=8,1000 sep={}]{1e\pomdpplot@index}}}%
      \fi
      \def\pomdpplot@separator{,}%
    }%
  \else
    \pomdpplot@get\pomdpplot@series{0}{#1_tick_step}\pomdpplot@tickstep
    \pomdpplot@get\pomdpplot@series{0}{#1_tick_count}\pomdpplot@tickcount
    \pgfplotsforeachungrouped \pomdpplot@index in {0,...,\pomdpplot@tickcount}{%
      \pgfmathsetmacro\pomdpplot@tick{\pomdpplot@index*\pomdpplot@tickstep}%
      \edef\pomdpplot@ticks{\pomdpplot@ticks\pomdpplot@separator\pomdpplot@tick}%
      \edef\pomdpplot@ticklabels{\pomdpplot@ticklabels\pomdpplot@separator
        {\noexpand\pgfmathprintnumber[fixed,precision=2]{\pomdpplot@tick}}}%
      \def\pomdpplot@separator{,}%
    }%
  \fi
  \pomdpplot@failuretick{memout}{MO}%
  \pomdpplot@failuretick{timeout}{TO}%
  \pomdpplot@failuretick{failed}{Failed}%
  \edef\pomdpplot@command{\noexpand\pgfplotsset{pomdp aggregate failure axis/.style={%
    /pgfplots/.cd,ymin=\pomdpplot@ymin,ymax=\pomdpplot@ymax,
    enlarge y limits=false,scaled y ticks=false,
    ytick={\pomdpplot@ticks},yticklabels={\unexpanded\expandafter{\pomdpplot@ticklabels}},
    extra y ticks={\pomdpplot@extraticks},extra y tick labels={},
    extra y tick style={grid=major,tick style={draw=none},
      major grid style={solid,draw=black!65,line width=.4pt}}}}}%
  \pomdpplot@command
}
\newcommand{\pomdpplot@allcomparisonplot}[2]{%
  \begingroup
  \pomdpplot@showinstancelegendtrue
  \def\pomdpplot@ymode{log}\def\pomdpplot@plotstyle{only marks}\def\pomdpplot@axisstyle{}%
  \def\pomdpplot@ylabel{time (s)}%
  \ifnum#2=0\relax
    \def\pomdpplot@xlabel{Size limit $\tau$}%
    \def\pomdpplot@title{(a) Runtime}%
    \pomdpplot@failureaxis{runtime}\def\pomdpplot@axisstyle{pomdp aggregate failure axis}%
  \else
    \ifnum#2=1\relax
      \def\pomdpplot@metric{under_gap}\def\pomdpplot@title{(b) Under-approximation}%
    \else
      \def\pomdpplot@metric{over_gap}\def\pomdpplot@title{(c) Over-approximation}%
    \fi
    \def\pomdpplot@xlabel{Size limit $\tau$}%
    \pomdpplot@get\pomdpplot@series{0}{\pomdpplot@metric _ylabel}\pomdpplot@ylabel
    \def\pomdpplot@ymode{linear}%
    \edef\pomdpplot@command{\noexpand\pomdpplot@failureaxis{\pomdpplot@metric}}\pomdpplot@command
    \def\pomdpplot@axisstyle{pomdp aggregate failure axis}%
  \fi
  \begin{tikzpicture}\begin{axis}[
    pomdp aggregate axis,xmode=log,ymode=\pomdpplot@ymode,
    enlarge x limits=.03,enlarge y limits=.06,
    title={\pomdpplot@title},xlabel={\pomdpplot@xlabel},ylabel={\pomdpplot@ylabel},
    legend columns=4,legend style={at={(.5,-.25)},anchor=north,
      cells={inner ysep=.5pt}},\pomdpplot@axisstyle,#1]
    \pgfplotsforeachungrouped\pomdpplot@row in {0,...,\pomdpplot@lastseries}{%
      \pomdpplot@allpalette
      \pomdpplot@get\pomdpplot@series{\pomdpplot@row}{aggregate_label}\pomdpplot@label
      \edef\pomdpplot@command{\noexpand\addplot[color=\pomdpplot@colour,mark=\pomdpplot@mark,
        mark size=.7pt,line width=.6pt,mark options={solid},
        \pomdpplot@plotstyle]}\pomdpplot@command
        table[x=threshold,y=\pomdpplot@failuremetric _\pomdpplot@seriesindex]{\pomdpplot@curves};
      \ifpomdpplot@showinstancelegend
        \edef\pomdpplot@command{\noexpand\addlegendentry{\unexpanded\expandafter{\pomdpplot@label}}}%
        \pomdpplot@command
      \fi
    }%
  \end{axis}\end{tikzpicture}\endgroup
}

% Aligned legend columns; prefer five columns, then four or three if needed.
\newcommand{\pomdpplot@legendentry}[1]{%
  \begingroup
    \edef\pomdpplot@row{#1}\pomdpplot@allpalette
    \pomdpplot@get\pomdpplot@series{#1}{aggregate_label}\pomdpplot@label
    \begin{tikzpicture}[baseline=-.5ex]
      \path[use as bounding box] (-1mm,-.7mm) rectangle (1mm,.7mm);
      \draw[color=\pomdpplot@colour,mark=\pomdpplot@mark,
        mark size=.7pt,line width=.6pt,mark options={solid}]
        plot[only marks] coordinates {(0,0)};
    \end{tikzpicture}\hspace{.5mm}\pomdpplot@label
  \endgroup
}
\newcommand{\pomdpplot@legendrows}[1]{%
  \def\pomdpplot@legendentries{}\def\pomdpplot@separator{}%
  \pgfplotsforeachungrouped\pomdpplot@row in {0,...,\pomdpplot@lastseries}{%
    \edef\pomdpplot@legendentries{%
      \unexpanded\expandafter{\pomdpplot@legendentries}%
      \unexpanded\expandafter{\pomdpplot@separator}%
      \noexpand\pomdpplot@legendentry{\pomdpplot@row}}%
    \pgfmathtruncatemacro\pomdpplot@column{mod(\pomdpplot@row+1,#1)}%
    \ifnum\pomdpplot@column=0\relax
      \def\pomdpplot@separator{\\}%
    \else
      \def\pomdpplot@separator{&}%
    \fi
  }%
}
\newcommand{\pomdpplot@measurelegend}[1]{%
  \def\pomdpplot@legendcolumns{#1}\pomdpplot@legendrows{#1}%
  \sbox{\pomdpplot@legendbox}{\begin{tabular}{@{}*{#1}{l}@{}}%
    \pomdpplot@legendentries
  \end{tabular}}%
}
\newcommand{\pomdpplot@legend}{%
  \begingroup
    \normalfont\fontsize{6}{7}\selectfont
    \setlength{\fboxsep}{.6mm}\setlength{\fboxrule}{.4pt}%
    \setlength{\tabcolsep}{.5mm}%
    \pomdpplot@legendwidth=\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax
    \pomdpplot@measurelegend{5}%
    \ifdim\wd\pomdpplot@legendbox>\pomdpplot@legendwidth
      \pomdpplot@measurelegend{4}%
      \ifdim\wd\pomdpplot@legendbox>\pomdpplot@legendwidth
        \pomdpplot@measurelegend{3}%
      \fi
    \fi
    \typeout{POMDP legend columns: \pomdpplot@legendcolumns}%
    \fbox{\begin{tabular}{@{}*{\pomdpplot@legendcolumns}{l}@{}}%
      \pomdpplot@legendentries
    \end{tabular}}%
  \endgroup
}

\newcommand{\StormPOMDPAggregateFigure}{%
  \begin{minipage}{\linewidth}\centering
    \setlength{\tabcolsep}{.5mm}%
    \sbox{\pomdpplot@figurebox}{\begin{tabular}{@{}ccc@{}}
      \StormPOMDPAllRuntimeThresholdPlot[scale only axis,width=2.45cm,height=2.1cm,
        pomdp aggregate no instance legend] &
      \StormPOMDPAllUnderBoundThresholdPlot[scale only axis,width=2.45cm,height=2.1cm,
        pomdp aggregate no instance legend] &
      \StormPOMDPAllOverBoundThresholdPlot[scale only axis,width=2.45cm,height=2.1cm,
        pomdp aggregate no instance legend]
    \end{tabular}}%
    \typeout{POMDP panel row width: \the\wd\pomdpplot@figurebox; available: \the\linewidth}%
    \ifdim\wd\pomdpplot@figurebox>\linewidth
      \PackageError{pomdp-aggregate-plots}{Panel row exceeds the available width}
        {Use the full LNCS text width for the aggregate figure.}%
    \fi
    \usebox{\pomdpplot@figurebox}\par\vspace{1mm}
    \pomdpplot@legend\par
  \end{minipage}%
}
